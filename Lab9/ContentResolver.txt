Public class MainActivity extends AppCompatActivity  {
TextView txt;
@Override
Protected void onCreate(Bundle savedInstanceState) {
Super.onCreate(savedInstanceState);
setContentView(R.layout.activity_main);
txt = findViewByID(R.id.txtView);
//URI is for connecting contact database
Uri uri = ContactsContract.CommonDataKInds.Phone.Content_uri;
//Projection is used for fetecting datatype
String[] projection = new String[]{
ContactsContract.CommonDataKinds.Phone.DISPLAY_NAME_PRIMARY,
ContactsContract.CommonDataKinds,Phone.NUMBER,
};
//Selection is for selecting Which data from fields.
String selection = ContactsContact.CommonDataKinds.Phone.DISPLAY_NAME_PRIMARY + “=’hardik’”;
//Resolver is to fetch data from ContentProvider
ContentResolver  resolver = getContentResolver();
//Resolver.qurey user 5 parameter to make operation on data
Cursor res = resolver.qurey(uri,projection,selection,selectArgs: null,sortorder: null);
If(res.getCount()==0)
{
	Toast.makeText(context: this,text:”Records Not Found”,Toast.Length_SHORT).Show();
} Else {
	//Cursor.moveToNext() is used To move cursor
	While(res.novaToNext()){
	//Cursor.getString(0) is used to get Display_Name_Primary
	//Cursor.getString(1) is used to get number
	Txt.setText(res.getString(columnindex: 0)+”: Phone=”+ res.getString(columnindex:1));

}
}
catch (Exception e)
{
	Toast.makeText(context:this,txt:e.get,Toast.LENGTH_SHORT).Show();
}

}
}
